cmake_minimum_required(VERSION 3.16)
project(bridge_cpp_s)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(racs2_msg REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(LWS REQUIRED IMPORTED_TARGET libwebsockets)
get_filename_component(RACS2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_executable(bridge_cpp_s_node src/bridge_cpp_s_node.cpp)
target_compile_features(bridge_cpp_s_node PRIVATE cxx_std_17)
target_compile_options(bridge_cpp_s_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(bridge_cpp_s_node PRIVATE "${RACS2_ROOT}/Common/include")
ament_target_dependencies(bridge_cpp_s_node rclcpp racs2_msg)
target_link_libraries(bridge_cpp_s_node PkgConfig::LWS Threads::Threads)
install(TARGETS bridge_cpp_s_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
